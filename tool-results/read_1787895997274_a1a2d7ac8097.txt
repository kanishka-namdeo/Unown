    54→function resolvePromptInput(input: string | undefined, description: string): string | undefined {
    55→        if (!input) {
    56→                return undefined;
    57→        }
    58→
    59→        if (existsSync(input)) {
    60→                try {
    61→                        return stripBom(readFileSync(input, "utf-8"));
    62→                } catch (error) {
    63→                        console.error(chalk.yellow(`Warning: Could not read ${description} file ${input}: ${error}`));
    64→                        return input;
    65→                }
    66→        }
    67→
    68→        return input;
    69→}

    71→function loadContextFileFromDir(dir: string): { path: string; content: string } | null {
    72→        const candidates = ["AGENTS.override.md", "AGENTS.md", "AGENTS.MD", "CLAUDE.md", "CLAUDE.MD"];
    73→        for (const filename of candidates) {
    74→                const filePath = join(dir, filename);
    75→                if (existsSync(filePath)) {
    76→                        try {
    77→                                if (!statSync(filePath).isFile()) {
    78→                                        continue;
    79→                                }
    80→                                return {
    81→                                        path: filePath,
    82→                                        content: stripBom(readFileSync(filePath, "utf-8")),
    83→                                };
    84→                        } catch (error) {
    85→                                console.error(chalk.yellow(`Warning: Could not read ${filePath}: ${error}`));
    86→                        }
    87→                }
    88→        }
    89→        return null;
    90→}

    92→/**
    93→ * The main repo's context file that a nested linked worktree's own copy shadows: both
    94→ * occupy the same logical repository scope, so loading both applies that context twice. Returns
    95→ * undefined when nothing is shadowed, leaving normal ancestor inheritance alone.
    96→ *
    97→ * Returned canonicalized (realpath), because `git worktree add` writes the `.git`
    98→ * file's `gitdir:` target in realpath form while cwd may still be symlinked
    99→ * (macOS `/tmp` -> `/private/tmp`).
   100→ */
   101→function findShadowedContextFile(cwd: string): string | undefined {
   102→        const gitPaths = findGitPaths(cwd);
   103→        if (!gitPaths) return undefined;
   104→        const commonGitDir = canonicalizePath(gitPaths.commonGitDir);
   105→        const worktreeRoot = canonicalizePath(gitPaths.repoDir);
   106→        const mainRepoRoot = dirname(commonGitDir);
   107→        // False for an ordinary repo, where the two are the same dir, and for a sibling
   108→        // worktree (`git worktree add ../feat`), whose main repo is not an ancestor.
   109→        if (!worktreeRoot.startsWith(`${mainRepoRoot}${sep}`)) return undefined;
   110→        // dirname of the common git dir is the main worktree root only when that dir is
   111→        // itself checked out from the same repo. In a bare layout (`proj/.bare` +
   112→        // `proj/main`) it is just the directory holding `.bare`, which tracks nothing; a
   113→        // submodule's gitdir has no `commondir`, so it lands under `.git/modules`.
   114→        if (canonicalizePath(join(mainRepoRoot, ".git")) !== commonGitDir) return undefined;
   115→        const worktreeContextFile = loadContextFileFromDir(worktreeRoot);
   116→        return worktreeContextFile ? join(mainRepoRoot, basename(worktreeContextFile.path)) : undefined;
   117→}

   119→export function loadProjectContextFiles(options: {
   120→        cwd: string;
   121→        agentDir: string;
   122→}): Array<{ path: string; content: string }> {
   123→        const resolvedCwd = resolvePath(options.cwd);
   124→        const resolvedAgentDir = resolvePath(options.agentDir);
   125→
   126→        const contextFiles: Array<{ path: string; content: string }> = [];
   127→        const seenPaths = new Set<string>();
   128→
   129→        const globalContext = loadContextFileFromDir(resolvedAgentDir);
   130→        if (globalContext) {
   131→                contextFiles.push(globalContext);
   132→                seenPaths.add(globalContext.path);
   133→        }
   134→
   135→        const ancestorContextFiles: Array<{ path: string; content: string }> = [];
   136→
   137→        const shadowedContextFile = findShadowedContextFile(resolvedCwd);
   138→        let currentDir = resolvedCwd;
   139→
   140→        while (true) {
   141→                const contextFile = loadContextFileFromDir(currentDir);
   142→                const isShadowed =
   143→                        shadowedContextFile !== undefined && canonicalizePath(contextFile?.path ?? "") === shadowedContextFile;
   144→                if (contextFile && !isShadowed && !seenPaths.has(contextFile.path)) {
   145→                        ancestorContextFiles.unshift(contextFile);
   146→                        seenPaths.add(contextFile.path);
   147→                }
   148→
   149→                const parentDir = dirname(currentDir);
   150→                if (parentDir === currentDir) break;
   151→                currentDir = parentDir;
   152→        }
   153→
   154→        contextFiles.push(...ancestorContextFiles);
   155→
   156→        return contextFiles;
   157→}

   388→        async reload(options?: ResourceLoaderReloadOptions): Promise<void> {
   389→                resetTimings("extensions");
   390→
   391→                if (this.loaded) {
   392→                        clearExtensionCache();
   393→                }
   394→
   395→                let preTrustExtensions: LoadExtensionsResult | undefined;
   396→                if (options?.resolveProjectTrust) {
   397→                        preTrustExtensions = await this.loadProjectTrustExtensions();
   398→                        const projectTrusted = await options.resolveProjectTrust({ extensionsResult: preTrustExtensions });
   399→                        this.settingsManager.setProjectTrusted(projectTrusted);
   400→                }
   401→
   402→                // reload() preserves SettingsManager.projectTrusted and reloads settings for that trust state.
   403→                await this.settingsManager.reload();
   404→                const resolvedPaths = await this.packageManager.resolve();
   405→                const cliExtensionPaths = await this.packageManager.resolveExtensionSources(this.additionalExtensionPaths, {
   406→                        temporary: true,
   407→                });
   408→                // Kept on the instance so post-reload passes (extendResources) can still resolve package metadata.
   409→                this.resourceMetadataByPath = new Map();
   410→                const metadataByPath = this.resourceMetadataByPath;
   411→
   412→                this.extensionSkillSourceInfos = new Map();
   413→                this.extensionPromptSourceInfos = new Map();
   414→                this.extensionThemeSourceInfos = new Map();
   415→
   416→                // Helper to extract enabled paths and store metadata
   417→                const getEnabledResources = (resources: ResolvedResource[]): ResolvedResource[] => {
   418→                        for (const r of resources) {
   419→                                if (!metadataByPath.has(r.path)) {
   420→                                        metadataByPath.set(r.path, r.metadata);
   421→                                }
   422→                        }
   423→                        return resources.filter((r) => r.enabled);
   424→                };
   425→
   426→                const getEnabledPaths = (resources: ResolvedResource[]): string[] =>
   427→                        getEnabledResources(resources).map((r) => r.path);
   428→                const enabledExtensions = getEnabledPaths(resolvedPaths.extensions);
   429→                const enabledSkillResources = getEnabledResources(resolvedPaths.skills);
   430→                const enabledPrompts = getEnabledPaths(resolvedPaths.prompts);
   431→                const enabledThemes = getEnabledPaths(resolvedPaths.themes);
   432→
   433→                const enabledSkills = enabledSkillResources.map((resource) => this.mapSkillPath(resource, metadataByPath));
   434→
   435→                // Add CLI paths metadata
   436→                for (const r of cliExtensionPaths.extensions) {
   437→                        if (!metadataByPath.has(r.path)) {
   438→                                metadataByPath.set(r.path, { source: "cli", scope: "temporary", origin: "top-level" });
   439→                        }
   440→                }
   441→                for (const r of cliExtensionPaths.skills) {
   442→                        if (!metadataByPath.has(r.path)) {
   443→                                metadataByPath.set(r.path, { source: "cli", scope: "temporary", origin: "top-level" });
   444→                        }
   445→                }
   446→
   447→                const cliEnabledExtensions = getEnabledPaths(cliExtensionPaths.extensions);
   448→                const cliEnabledSkills = getEnabledPaths(cliExtensionPaths.skills);
   449→                const cliEnabledPrompts = getEnabledPaths(cliExtensionPaths.prompts);
   450→                const cliEnabledThemes = getEnabledPaths(cliExtensionPaths.themes);
   451→
   452→                const extensionPaths = this.noExtensions
   453→                        ? cliEnabledExtensions
   454→                        : this.mergePaths(cliEnabledExtensions, enabledExtensions);
   455→
   456→                const extensionsResult = await this.loadFinalExtensionSet(extensionPaths, preTrustExtensions);
   457→                for (const p of this.additionalExtensionPaths) {
   458→                        if (isLocalPath(p)) {
   459→                                const resolved = this.resolveResourcePath(p);
   460→                                if (!existsSync(resolved)) {
   461→                                        extensionsResult.errors.push({ path: resolved, error: `Extension path does not exist: ${resolved}` });
   462→                                }
   463→                        }
   464→                }
   465→                this.extensionsResult = this.extensionsOverride ? this.extensionsOverride(extensionsResult) : extensionsResult;
   466→                this.applyExtensionSourceInfo(this.extensionsResult.extensions, metadataByPath);
   467→
   468→                const skillPaths = this.noSkills
   469→                        ? this.mergePaths(cliEnabledSkills, this.additionalSkillPaths)
   470→                        : this.mergePaths([...cliEnabledSkills, ...enabledSkills], this.additionalSkillPaths);
   471→
   472→                this.lastSkillPaths = skillPaths;
   473→                this.updateSkillsFromPaths(skillPaths, metadataByPath);
   474→                for (const p of this.additionalSkillPaths) {
   475→                        if (isLocalPath(p)) {
   476→                                const resolved = this.resolveResourcePath(p);
   477→                                if (!existsSync(resolved) && !this.skillDiagnostics.some((d) => d.path === resolved)) {
   478→                                        this.skillDiagnostics.push({ type: "error", message: "Skill path does not exist", path: resolved });
   479→                                }
   480→                        }
   481→                }
   482→
   483→                const promptPaths = this.noPromptTemplates
   484→                        ? this.mergePaths(cliEnabledPrompts, this.additionalPromptTemplatePaths)
   485→                        : this.mergePaths([...cliEnabledPrompts, ...enabledPrompts], this.additionalPromptTemplatePaths);
   486→
   487→                this.lastPromptPaths = promptPaths;
   488→                this.updatePromptsFromPaths(promptPaths, metadataByPath);
   489→                for (const p of this.additionalPromptTemplatePaths) {
   490→                        if (isLocalPath(p)) {
   491→                                const resolved = this.resolveResourcePath(p);
   492→                                if (!existsSync(resolved) && !this.promptDiagnostics.some((d) => d.path === resolved)) {
   493→                                        this.promptDiagnostics.push({
   494→                                                type: "error",
   495→                                                message: "Prompt template path does not exist",
   496→                                                path: resolved,
   497→                                        });
   498→                                }
   499→                        }
   500→                }
   501→
   502→                const themePaths = this.noThemes
   503→                        ? this.mergePaths(cliEnabledThemes, this.additionalThemePaths)
   504→                        : this.mergePaths([...cliEnabledThemes, ...enabledThemes], this.additionalThemePaths);
   505→
   506→                this.lastThemePaths = themePaths;
   507→                this.updateThemesFromPaths(themePaths, metadataByPath);
   508→                for (const p of this.additionalThemePaths) {
   509→                        const resolved = this.resolveResourcePath(p);
   510→                        if (!existsSync(resolved) && !this.themeDiagnostics.some((d) => d.path === resolved)) {
   511→                                this.themeDiagnostics.push({ type: "error", message: "Theme path does not exist", path: resolved });
   512→                        }
   513→                }
   514→
   515→                const agentsFiles = {
   516→                        agentsFiles: this.noContextFiles
   517→                                ? []
   518→                                : loadProjectContextFiles({
   519→                                                cwd: this.cwd,
   520→                                                agentDir: this.agentDir,
   521→                                        }),
   522→                };
   523→                const resolvedAgentsFiles = this.agentsFilesOverride ? this.agentsFilesOverride(agentsFiles) : agentsFiles;
   524→                this.agentsFiles = resolvedAgentsFiles.agentsFiles;
   525→
   526→                const systemPromptSource = this.systemPromptSource ?? this.discoverSystemPromptFile();

   574→        private async loadFinalExtensionSet(
   575→                extensionPaths: string[],
   576→                preTrustExtensions: LoadExtensionsResult | undefined,
   577→        ): Promise<LoadExtensionsResult> {
   578→                if (!preTrustExtensions) {
   579→                        const extensionsResult = await loadExtensionsCached(extensionPaths, this.cwd, this.eventBus);
   580→                        const inlineExtensions = await this.loadExtensionFactories(extensionsResult.runtime);
   581→                        extensionsResult.extensions.push(...inlineExtensions.extensions);
   582→                        extensionsResult.errors.push(...inlineExtensions.errors);
   583→                        this.addExtensionConflictDiagnostics(extensionsResult);
   584→                        return extensionsResult;
   585→                }
   586→
   587→                const preloadedByPath = new Map(
   588→                        preTrustExtensions.extensions
   589→                                .filter((extension) => !extension.path.startsWith("<inline:"))
   590→                                .map((extension) => [extension.resolvedPath, extension]),
   591→                );
   592→                const failedPreloadPaths = new Set(
   593→                        preTrustExtensions.errors.map((error) => this.resolveExtensionLoadPath(error.path)),
   594→                );
   595→                const remainingPaths = extensionPaths.filter((path) => {
   596→                        const resolvedPath = this.resolveExtensionLoadPath(path);
   597→                        return !preloadedByPath.has(resolvedPath) && !failedPreloadPaths.has(resolvedPath);
   598→                });
   599→                const remainingExtensions = await loadExtensionsCached(
   600→                        remainingPaths,
   601→                        this.cwd,
   602→                        this.eventBus,
   603→                        preTrustExtensions.runtime,
   604→                );
   605→                const loadedByPath = new Map(preloadedByPath);
   606→                for (const extension of remainingExtensions.extensions) {
   607→                        loadedByPath.set(extension.resolvedPath, extension);
   608→                }
   609→
   610→                const inlineExtensions = preTrustExtensions.extensions.filter((extension) =>
   611→                        extension.path.startsWith("<inline:"),
   612→                );
   613→                const orderedExtensions = extensionPaths
   614→                        .map((path) => loadedByPath.get(this.resolveExtensionLoadPath(path)))
   615→                        .filter((extension): extension is Extension => extension !== undefined);
   616→                orderedExtensions.push(...inlineExtensions);
   617→
   618→                const extensionsResult: LoadExtensionsResult = {
   619→                        extensions: orderedExtensions,
   620→                        errors: [...preTrustExtensions.errors, ...remainingExtensions.errors],
   621→                        runtime: preTrustExtensions.runtime,
   622→                };
   623→                this.addExtensionConflictDiagnostics(extensionsResult);
   624→                return extensionsResult;
   625→        }

   755→        private findSourceInfoForPath(
   756→                resourcePath: string,
   757→                extraSourceInfos?: Map<string, SourceInfo>,
   758→                metadataByPath?: Map<string, PathMetadata>,
   759→        ): SourceInfo | undefined {
   760→                if (!resourcePath) {
   761→                        return undefined;
   762→                }
   763→
   764→                if (resourcePath.startsWith("<")) {
   765→                        return this.getDefaultSourceInfoForPath(resourcePath);
   766→                }
   767→
   768→                const normalizedResourcePath = resolve(resourcePath);
   769→                if (extraSourceInfos) {
   770→                        for (const [sourcePath, sourceInfo] of extraSourceInfos.entries()) {
   771→                                const normalizedSourcePath = resolve(sourcePath);
   772→                                if (
   773→                                        normalizedResourcePath === normalizedSourcePath ||
   774→                                        normalizedResourcePath.startsWith(`${normalizedSourcePath}${sep}`)
   775→                                ) {
   776→                                        return { ...sourceInfo, path: resourcePath };
   777→                                }
   778→                        }
   779→                }
   780→
   781→                if (metadataByPath) {
   782→                        const exact = metadataByPath.get(normalizedResourcePath) ?? metadataByPath.get(resourcePath);
   783→                        if (exact) {
   784→                                return createSourceInfo(resourcePath, exact);
   785→                        }
   786→
   787→                        for (const [sourcePath, metadata] of metadataByPath.entries()) {
   788→                                const normalizedSourcePath = resolve(sourcePath);
   789→                                if (
   790→                                        normalizedResourcePath === normalizedSourcePath ||
   791→                                        normalizedResourcePath.startsWith(`${normalizedSourcePath}${sep}`)
   792→                                ) {
   793→                                        return createSourceInfo(resourcePath, metadata);
   794→                                }
   795→                        }
   796→                }
   797→
   798→                return undefined;
   799→        }

   801→        private getDefaultSourceInfoForPath(filePath: string): SourceInfo {
   802→                if (filePath.startsWith("<") && filePath.endsWith(">")) {
   803→                        return {
   804→                                path: filePath,
   805→                                source: filePath.slice(1, -1).split(":")[0] || "temporary",
   806→                                scope: "temporary",
   807→                                origin: "top-level",
   808→                        };
   809→                }
   810→
   811→                const normalizedPath = resolve(filePath);
   812→                const agentRoots = [
   813→                        join(this.agentDir, "skills"),
   814→                        join(this.agentDir, "prompts"),
   815→                        join(this.agentDir, "themes"),
   816→                        join(this.agentDir, "extensions"),
   817→                ];
   818→                const projectRoots = [
   819→                        join(this.cwd, CONFIG_DIR_NAME, "skills"),
   820→                        join(this.cwd, CONFIG_DIR_NAME, "prompts"),
   821→                        join(this.cwd, CONFIG_DIR_NAME, "themes"),
   822→                        join(this.cwd, CONFIG_DIR_NAME, "extensions"),
   823→                ];
   824→
   825→                for (const root of agentRoots) {
   826→                        if (this.isUnderPath(normalizedPath, root)) {
   827→                                return { path: filePath, source: "local", scope: "user", origin: "top-level", baseDir: root };
   828→                        }
   829→                }
   830→
   831→                for (const root of projectRoots) {
   832→                        if (this.isUnderPath(normalizedPath, root)) {
   833→                                return { path: filePath, source: "local", scope: "project", origin: "top-level", baseDir: root };
   834→                        }
   835→                }
   836→
   837→                return {
   838→                        path: filePath,
   839→                        source: "local",
   840→                        scope: "temporary",
   841→                        origin: "top-level",
   842→                        baseDir: statSync(normalizedPath).isDirectory() ? normalizedPath : resolve(normalizedPath, ".."),
   843→                };
   844→        }

   865→        private loadThemes(
   866→                paths: string[],
   867→                includeDefaults: boolean = true,
   868→        ): {
   869→                themes: Theme[];
   870→                diagnostics: ResourceDiagnostic[];
   871→        } {
   872→                const themes: Theme[] = [];
   873→                const diagnostics: ResourceDiagnostic[] = [];
   874→                if (includeDefaults) {
   875→                        const defaultDirs = [join(this.agentDir, "themes"), join(this.cwd, CONFIG_DIR_NAME, "themes")];
   876→
   877→                        for (const dir of defaultDirs) {
   878→                                this.loadThemesFromDir(dir, themes, diagnostics);
   879→                        }
   880→                }
   881→
   882→                for (const p of paths) {
   883→                        const resolved = this.resolveResourcePath(p);
   884→                        if (!existsSync(resolved)) {
   885→                                diagnostics.push({ type: "warning", message: "theme path does not exist", path: resolved });
   886→                                continue;
   887→                        }
   888→
   889→                        try {
   890→                                const stats = statSync(resolved);
   891→                                if (stats.isDirectory()) {
   892→                                        this.loadThemesFromDir(resolved, themes, diagnostics);
   893→                                } else if (stats.isFile() && resolved.endsWith(".json")) {
   894→                                        this.loadThemeFromFile(resolved, themes, diagnostics);
   895→                                } else {
   896→                                        diagnostics.push({ type: "warning", message: "theme path is not a json file", path: resolved });
   897→                                }
   898→                        } catch (error) {
   899→                                const message = error instanceof Error ? error.message : "failed to read theme path";
   900→                                diagnostics.push({ type: "warning", message, path: resolved });
   901→                        }
   902→                }
   903→
   904→                return { themes, diagnostics };
   905→        }

   907→        private loadThemesFromDir(dir: string, themes: Theme[], diagnostics: ResourceDiagnostic[]): void {
   908→                if (!existsSync(dir)) {
   909→                        return;
   910→                }
   911→
   912→                try {
   913→                        const entries = readdirSync(dir, { withFileTypes: true });
   914→                        for (const entry of entries) {
   915→                                let isFile = entry.isFile();
   916→                                if (entry.isSymbolicLink()) {
   917→                                        try {
   918→                                                isFile = statSync(join(dir, entry.name)).isFile();
   919→                                        } catch {
   920→                                                continue;
   921→                                        }
   922→                                }
   923→                                if (!isFile) {
   924→                                        continue;
   925→                                }
   926→                                if (!entry.name.endsWith(".json")) {
   927→                                        continue;
   928→                                }
   929→                                this.loadThemeFromFile(join(dir, entry.name), themes, diagnostics);
   930→                        }
   931→                } catch (error) {
   932→                        const message = error instanceof Error ? error.message : "failed to read theme directory";
   933→                        diagnostics.push({ type: "warning", message, path: dir });
   934→                }
   935→        }

  1060→        private detectExtensionConflicts(extensions: Extension[]): Array<{ path: string; message: string }> {
  1061→                const conflicts: Array<{ path: string; message: string }> = [];
  1062→
  1063→                // Track which extension registered each tool and flag
  1064→                const toolOwners = new Map<string, string>();
  1065→                const flagOwners = new Map<string, string>();
  1066→
  1067→                for (const ext of extensions) {
  1068→                        // Check tools
  1069→                        for (const toolName of ext.tools.keys()) {
  1070→                                const existingOwner = toolOwners.get(toolName);
  1071→                                if (existingOwner && existingOwner !== ext.path) {
  1072→                                        conflicts.push({
  1073→                                                path: ext.path,
  1074→                                                message: `Tool "${toolName}" conflicts with ${existingOwner}`,
  1075→                                        });
  1076→                                } else {
  1077→                                        toolOwners.set(toolName, ext.path);
  1078→                                }
  1079→                        }
  1080→
