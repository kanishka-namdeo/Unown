   252→         *     }
   253→         *     super.handleInput(data);  // App keybindings + text editing

   448→/**
   449→ * Tool definition for registerTool().
   450→ */
   451→export interface ToolDefinition<TParams extends TSchema = TSchema, TDetails = unknown, TState = any> {
   452→        /** Tool name (used in LLM tool calls) */
   453→        name: string;
   454→        /** Human-readable label for UI */
   455→        label: string;
   456→        /** Description for LLM */
   457→        description: string;
   458→        /** Optional one-line snippet for the Available tools section in the default system prompt. Custom tools are omitted from that section when this is not provided. */
   459→        promptSnippet?: string;
   460→        /** Optional guideline bullets appended to the default system prompt Guidelines section when this tool is active. */
   461→        promptGuidelines?: string[];
   462→        /** Parameter schema (TypeBox) */
   463→        parameters: TParams;
   464→        /** Optional provider-side constrained sampling request for this tool. Set false to explicitly disable it, equivalent to leaving it undefined. */
   465→        constrainedSampling?: false | ConstrainedSamplingConfig;
   466→        /** Controls whether ToolExecutionComponent renders the standard colored shell or the tool renders its own framing. */
   467→        renderShell?: "default" | "self";
   468→
   469→        /** Optional compatibility shim to prepare raw tool call arguments before schema validation. Must return an object conforming to TParams. */
   470→        prepareArguments?: (args: unknown) => Static<TParams>;
   471→
   472→        /**
   473→         * Per-tool execution mode override.
   474→         * - "sequential": this tool must execute one at a time with other tool calls.
   475→         * - "parallel": this tool can execute concurrently with other tool calls.
   476→         *
   477→         * If omitted, the default execution mode applies.
   478→         */
   479→        executionMode?: ToolExecutionMode;
   480→
   481→        /** Execute the tool. */
   482→        execute(
   483→                toolCallId: string,
   484→                params: Static<TParams>,
   485→                signal: AbortSignal | undefined,
   486→                onUpdate: AgentToolUpdateCallback<TDetails> | undefined,
   487→                ctx: ExtensionContext,
   488→        ): Promise<AgentToolResult<TDetails>>;
   489→
   490→        /** Custom rendering for tool call display */
   491→        renderCall?: (args: Static<TParams>, theme: Theme, context: ToolRenderContext<TState, Static<TParams>>) => Component;
   492→
   493→        /** Custom rendering for tool result display */
   494→        renderResult?: (
   495→                result: AgentToolResult<TDetails>,
   496→                options: ToolRenderResultOptions,
   497→                theme: Theme,
   498→                context: ToolRenderContext<TState, Static<TParams>>,
   499→        ) => Component;
   500→}
   501→
   502→type AnyToolDefinition = ToolDefinition<any, any, any>;
   503→
   504→/**
   505→ * Preserve parameter inference for standalone tool definitions.
   506→ *
   507→ * Use this when assigning a tool to a variable or passing it through arrays such
   508→ * as `customTools`, where contextual typing would otherwise widen params to
   509→ * `unknown`.
   510→ */
   511→export function defineTool<TParams extends TSchema, TDetails = unknown, TState = any>(
   512→        tool: ToolDefinition<TParams, TDetails, TState>,
   513→): ToolDefinition<TParams, TDetails, TState> & AnyToolDefinition {
   514→        return tool as ToolDefinition<TParams, TDetails, TState> & AnyToolDefinition;
   515→}

  1049→/**

  1051→ *
  1052→ * Built-in tools narrow automatically (no type params needed):
  1053→ * ```ts
  1054→ * if (isToolCallEventType("bash", event)) {
  1055→ *   event.input.command;  // string
  1056→ * }
  1057→ * ```
  1058→ *
  1059→ * Custom tools require explicit type parameters:
  1060→ * ```ts
  1061→ * if (isToolCallEventType<"my_tool", MyToolInput>("my_tool", event)) {
  1062→ *   event.input.action;  // typed
  1063→ * }
  1064→ * ```
  1065→ *
  1066→ * Note: Direct narrowing via `event.toolName === "bash"` doesn't work because
  1067→ * CustomToolCallEvent.toolName is `string` which overlaps with all literals.
  1068→ */

  1070→export function isToolCallEventType(toolName: "powershell", event: ToolCallEvent): event is PowerShellToolCallEvent;

  1072→export function isToolCallEventType(toolName: "edit", event: ToolCallEvent): event is EditToolCallEvent;
  1073→export function isToolCallEventType(toolName: "write", event: ToolCallEvent): event is WriteToolCallEvent;

  1075→export function isToolCallEventType(toolName: "find", event: ToolCallEvent): event is FindToolCallEvent;
  1076→export function isToolCallEventType(toolName: "ls", event: ToolCallEvent): event is LsToolCallEvent;
  1077→export function isToolCallEventType<TName extends string, TInput extends Record<string, unknown>>(
  1078→        toolName: TName,

  1456→         *     }
  1457→         *   ]
  1458→         * });
  1459→         *
  1460→         * @example
  1461→         * // Override baseUrl for an existing provider
  1462→         * pi.registerProvider("anthropic", {
  1463→         *   baseUrl: "https://proxy.example.com"
  1464→         * });
  1465→         *
  1466→         * @example
